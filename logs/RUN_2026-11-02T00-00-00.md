```python
DEFAULT_SPEED = 1.0
MIN_SPEED = 0.25
MAX_SPEED = 3.0
MIN_DELAY = 0.05


def _clamp_speed(raw_value: Any, log: Callable[..., Any] = print) -> float:
    try:
        value = float(raw_value)
    except (TypeError, ValueError):
        _log(log, f"speed invalid={raw_value!r}; fallback to {DEFAULT_SPEED:.2f}")
        return DEFAULT_SPEED
    if value < MIN_SPEED:
        _log(log, f"speed clamped from {value:.2f} to {MIN_SPEED:.2f}")
        return MIN_SPEED
    if value > MAX_SPEED:
        _log(log, f"speed clamped from {value:.2f} to {MAX_SPEED:.2f}")
        return MAX_SPEED
    return value


def _settings(project_dir: str | Path, log: Callable[..., Any] = print) -> dict[str, Any]:
    nav, _ = _read_nav(project_dir)
    settings = nav.get("settings", {}) if isinstance(nav.get("settings"), dict) else {}
    return {
        "click_radius_px": int(settings.get("click_radius_px", DEFAULT_CLICK_RADIUS_PX) or DEFAULT_CLICK_RADIUS_PX),
        "approach_timeout_s": float(settings.get("approach_timeout_s", DEFAULT_APPROACH_TIMEOUT_S) or DEFAULT_APPROACH_TIMEOUT_S),
        "hold_chunk_ms": int(settings.get("hold_chunk_ms", DEFAULT_HOLD_CHUNK_MS) or DEFAULT_HOLD_CHUNK_MS),
        "frame_interval_ms": int(settings.get("frame_interval_ms", DEFAULT_FRAME_INTERVAL_MS) or DEFAULT_FRAME_INTERVAL_MS),
        "match_threshold": _sanitize_threshold(settings.get("match_threshold", DEFAULT_MATCH_THRESHOLD), log=log),
        "speed": _clamp_speed(settings.get("speed", settings.get("movement_speed", DEFAULT_SPEED)), log=log),
    }


def _scaled_delay(project_dir: str | Path, base: float, log: Callable[..., Any] = print) -> float:
    settings = _settings(project_dir, log=log)
    speed = _clamp_speed(settings.get("speed", DEFAULT_SPEED), log=log)
    return max(MIN_DELAY, float(base) / speed)
```

### Отклонённые правки с причинами
1. Отклонён вариант патча `gui_base.py` для `_poll_app_log_queue()`, который полностью удаляет чтение `self.app.log_queue`.
   - Причина: в текущем `main` `gui.py` продолжает писать в `GuiApp.log_queue` через `GuiApp.log()`, поэтому такой патч ломает вывод GUI-логов в консоли вкладки настроек.
2. Не подтверждено автоприменение полного атомарного набора в `main`.
   - Причина: по read-only проверке соответствующие изменения в `main` отсутствуют; они должны применяться вручную и проверяться отдельно.
3. Не принят вывод, что открытый PR №1 автоматически закрывает весь список проблем круга 2.
   - Причина: PR открыт, не влит и по описанию покрывает только часть идей; он не заменяет ручную проверку и не подтверждает внесение всех патчей в `main`.
