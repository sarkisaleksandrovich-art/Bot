```python
def _scaled_delay(base: float, settings: dict[str, Any]) -> float:
    try:
        speed = float(settings.get("speed", DEFAULT_SPEED) or DEFAULT_SPEED)
    except (TypeError, ValueError):
        speed = DEFAULT_SPEED
    if speed <= 0:
        speed = DEFAULT_SPEED
    return max(MIN_DELAY, float(base) / speed)


def _capture_frame(runtime: NavigationRuntime, log: Callable[..., Any]) -> tuple[Any, bool]:

def _normalize_roi(roi: Any, frame_width: int, frame_height: int) -> tuple[int, int, int, int]:
    if not isinstance(roi, (list, tuple)) or len(roi) != 4:
        return (0, 0, frame_width, frame_height)
    try:
        x1, y1, x2, y2 = (int(roi[0]), int(roi[1]), int(roi[2]), int(roi[3]))
    except Exception:
        return (0, 0, frame_width, frame_height)
    x1 = max(0, min(frame_width, x1))
    y1 = max(0, min(frame_height, y1))
    x2 = max(x1 + 1, min(frame_width, x2))
    y2 = max(y1 + 1, min(frame_height, y2))
    return (x1, y1, x2, y2)


def _template_roi(template: Any, frame_width: int, frame_height: int) -> tuple[int, int, int, int]:
    if isinstance(template, dict):
        return _normalize_roi(template.get("roi"), frame_width, frame_height)
    return (0, 0, frame_width, frame_height)


def _wait_for_template(
    project_dir: str | Path,
    template: Any,
    timeout_s: float,
    log: Callable[..., Any] = print,
    stop_event: Any = None,
    settings: Optional[dict[str, Any]] = None,
) -> tuple[bool, tuple[bool, int, int, int, float], Any]:
    runtime = _build_runtime(project_dir, log=log)
    local_settings = settings or _settings(project_dir, log=log)
    deadline = time.monotonic() + float(timeout_s)
    last_result: tuple[bool, int, int, int, float] = (False, 0, 0, 0, 0.0)
    last_frame: Any = None
    while time.monotonic() < deadline:
        if stop_event is not None and stop_event.is_set():
            return False, last_result, last_frame
        frame, _ = _capture_frame(runtime, log)
        last_frame = frame
        last_result = find_template(project_dir, template, frame=frame, settings=local_settings)
        if last_result[0]:
            return True, last_result, frame
        time.sleep(_scaled_delay(local_settings["frame_interval_ms"] / 1000.0, local_settings))
    return False, last_result, last_frame
```
